export function getStoredApiKey(): string | null {
  if (typeof window === 'undefined') return null;
  return localStorage.getItem(GLADIA_API_KEY_STORAGE);
}

export function clearStoredApiKey(): void {
  if (typeof window === 'undefined') return;
  localStorage.removeItem(GLADIA_API_KEY_STORAGE);
}

export function createGladiaClient(apiKey: string) {
  // Use CDN version (v0.5.2 from unpkg, same as official sample)
  if (typeof window !== 'undefined' && (window as any).Gladia) {
    try {
      const client = new (window as any).Gladia.GladiaClient({ apiKey });
      return client;
    } catch (error) {
      console.error('Error creating Gladia client:', error);
      throw error;
    }
  }
