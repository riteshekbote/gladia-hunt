function AppShell({
  id,
  gate,
  activeScreen = "home",
  children,
}: {
  id: string;
  gate?: boolean;
  activeScreen?: NavScreen;
  children: React.ReactNode;
}) {
  return (
    <div
      id={id}
      data-screenshot={id}
      className={`layout${gate ? " layout-gate" : ""}`}

      </main>
    </div>
  );
}
